import datetime
import hashlib
import json
import os
import pathlib
import subprocess
import sys

root = pathlib.Path('/workspace/FsusBlog')
out = pathlib.Path('/tmp/fsusblog-eight981')
side = sys.argv[1]
manifest = json.loads((out / 'source-manifest.json').read_text())
env = dict(os.environ, FSUSBLOG_USE_LOCAL_FSUSUI='1', FSUSBLOG_LOCAL_FSUSUI_ROOT='/workspace/FsusUI', NODE_OPTIONS='--max-old-space-size=8192')
checks = [
    ('resolver', ['node', 'src/frontend/scripts/fsusui-root.mjs', '--repository-root', str(root), '--override', '/workspace/FsusUI']),
    ('application', ['node', 'src/frontend/scripts/run-local-fsusui-typecheck.mjs', '--application']),
    ('effective-config', ['node', 'src/frontend/node_modules/vue-tsc/bin/vue-tsc.js', '--showConfig', '-p', 'src/frontend/.tmp/typecheck/tsconfig.fsusui-local.app.generated.json']),
    ('node-types', ['node', 'src/frontend/node_modules/typescript/bin/tsc', '--noEmit', '-p', 'src/frontend/tsconfig.node.json']),
    ('full-style', ['node', 'tools/frontend/check-frontend-style.mjs', '--mode', 'full']),
    ('strict-full-style', ['node', 'tools/frontend/check-frontend-style.mjs', '--mode', 'strict-full']),
    ('strict-owned', ['node', 'tools/frontend/check-frontend-style.mjs', '--mode', 'strict-full'] + [arg for p in manifest['productionPaths'] for arg in ['--file', p]]),
    ('boundary', ['node', 'tests/frontend/checks/static/check-fsusui-boundary.mjs']),
    ('binary', ['node', 'tests/frontend/checks/static/check-binary-import-boundaries.mjs']),
    ('intent', ['node', 'tools/governance/check-code-intent.mjs']),
    ('lifecycle', ['node', 'tools/docs/generate-document-lifecycle.mjs', '--check']),
    ('authoritative', ['node', 'tools/docs/check-authoritative-development-docs.mjs']),
    ('source-whitespace', ['git', 'diff', '--check', manifest['base'], '--'] + manifest['productionPaths']),
    ('aggregate-whitespace', ['git', 'diff', '--cached', '--check', manifest['base']]),
]
records = []
for name, argv in checks:
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    log = out / (side + '-' + name + '.txt')
    with log.open('wb') as stream:
        result = subprocess.run(argv, cwd=root, env=env, stdout=stream, stderr=subprocess.STDOUT)
    data = log.read_bytes()
    (out / (side + '-' + name + '.exit')).write_text(str(result.returncode) + '\n')
    records.append(dict(name=name, argv=argv, cwd=str(root), environment={k: env[k] for k in ['FSUSBLOG_USE_LOCAL_FSUSUI','FSUSBLOG_LOCAL_FSUSUI_ROOT','NODE_OPTIONS']}, started=started, finished=datetime.datetime.now(datetime.timezone.utc).isoformat(), exit=result.returncode, log=log.name, bytes=len(data), lines=len(data.splitlines()), sha256=hashlib.sha256(data).hexdigest()))
    (out / (side + '-commands.json')).write_text(json.dumps(records, indent=4) + '\n')
    print(side, name, 'exit', result.returncode, 'bytes', len(data), flush=True)
generated = root / 'src/frontend/.tmp/typecheck/tsconfig.fsusui-local.app.generated.json'
if generated.exists():
    (out / (side + '-generated-config.snapshot.json')).write_bytes(generated.read_bytes())
