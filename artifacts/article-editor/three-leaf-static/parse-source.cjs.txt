const fs = require('node:fs');
const path = require('node:path');
const { createRequire } = require('node:module');
const load = createRequire(path.join(process.cwd(), 'src/frontend/package.json'));
const ts = load('typescript');
const sfc = load('@vue/compiler-sfc');
const dom = load('@vue/compiler-dom');
const inputs = JSON.parse(fs.readFileSync('/workspace/three-leaf-evidence/exact-source-inputs.json','utf8'));
const files = inputs.paths.map(row => row.path).filter(file => file.startsWith('src/frontend/'));
const results = [];
let errors = 0;
for (const file of files) {
    const source = fs.readFileSync(file,'utf8');
    let scripts = [{content:source}];
    if (file.endsWith('.vue')) {
        const parsed = sfc.parse(source,{filename:file});
        errors += parsed.errors.length;
        for (const error of parsed.errors) console.error(file,error);
        scripts = [parsed.descriptor.script,parsed.descriptor.scriptSetup].filter(Boolean);
        if (parsed.descriptor.template) dom.baseParse(parsed.descriptor.template.content,{onError(error){console.error(file,error.message);errors++;}});
    }
    for (const block of scripts) {
        const parsed = ts.createSourceFile(file+'.ts',block.content,ts.ScriptTarget.Latest,true,ts.ScriptKind.TS);
        errors += parsed.parseDiagnostics.length;
        for (const error of parsed.parseDiagnostics) console.error(file,ts.flattenDiagnosticMessageText(error.messageText,'\n'));
    }
    results.push({path:file,scriptBlocks:scripts.length});
}
console.log(JSON.stringify({kind:'source syntax parsing only; no product modules loaded, imported, evaluated or called',files:results,errors},null,2));
process.exitCode = errors ? 1 : 0;
