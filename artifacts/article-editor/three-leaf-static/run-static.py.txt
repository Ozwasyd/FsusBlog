import concurrent.futures
import datetime
import hashlib
import json
import os
from pathlib import Path
import subprocess

OUT = Path('/workspace/three-leaf-evidence')
ENV = os.environ.copy()
ENV.update({
    'PATH': '/workspace/.setup/tools/node_modules/.bin:/workspace/.setup/dotnet:' + ENV['PATH'],
    'DOTNET_CLI_HOME': '/workspace/.setup/dotnet-home',
    'NUGET_PACKAGES': '/workspace/.setup/nuget',
    'DOTNET_CLI_TELEMETRY_OPTOUT': '1',
    'FSUSBLOG_USE_LOCAL_FSUSUI': '1',
    'FSUSBLOG_LOCAL_FSUSUI_ROOT': '/workspace/FsusUI-three-leaf-ui973',
})
COMMANDS = [
    ('server-compile', ['dotnet', 'build', 'src/backend/FsusBlog.Server.csproj', '-c', 'Release', '--no-incremental', '/p:RestoreLockedMode=true', '/p:PublishClientAssets=false']),
    ('application', ['node', 'src/frontend/scripts/run-local-fsusui-typecheck.mjs', '--application']),
    ('full-csharp', ['node', 'tools/dotnet/check-csharp-style.mjs', '--mode', 'full']),
    ('full-style', ['node', 'tools/frontend/check-frontend-style.mjs', '--mode', 'full']),
    ('fsusui-boundary', ['node', 'tests/frontend/checks/static/check-fsusui-boundary.mjs']),
    ('binary-boundary', ['node', 'tests/frontend/checks/static/check-binary-import-boundaries.mjs']),
    ('code-intent', ['node', 'tools/governance/check-code-intent.mjs']),
    ('lifecycle', ['node', 'tools/docs/generate-document-lifecycle.mjs', '--check']),
    ('authoritative', ['node', 'tools/docs/check-authoritative-development-docs.mjs']),
    ('graph-factory', ['node', 'tools/governance/check-canonical-article-graph-factory.mjs']),
    ('strict-csharp', ['node', 'tools/dotnet/check-csharp-style.mjs', '--mode', 'strict-full', '--file', 'src/backend/Services/Authoring/ArticleGraphQueryService.cs']),
]
FRONTEND = [r['path'] for r in json.loads((OUT/'exact-source-inputs.json').read_text())['paths'] if r['path'].startswith('src/frontend/')]

def run(stage, name, argv):
    root = '/workspace/FsusBlog-three-leaf' + ('-base' if stage == 'base' else '')
    stem = OUT / (stage + '-' + name)
    record = {'cwd': root, 'argv': argv, 'envOverrides': {k: ENV[k] for k in ['PATH','DOTNET_CLI_HOME','NUGET_PACKAGES','DOTNET_CLI_TELEMETRY_OPTOUT','FSUSBLOG_USE_LOCAL_FSUSUI','FSUSBLOG_LOCAL_FSUSUI_ROOT']}, 'startedUTC': datetime.datetime.now(datetime.timezone.utc).isoformat()}
    with stem.with_suffix('.log').open('wb') as output:
        result = subprocess.run(argv, cwd=root, env=ENV, stdout=output, stderr=subprocess.STDOUT)
    data = stem.with_suffix('.log').read_bytes()
    record.update(exit=result.returncode, finishedUTC=datetime.datetime.now(datetime.timezone.utc).isoformat(), bytes=len(data), lines=len(data.splitlines()), sha256=hashlib.sha256(data).hexdigest())
    stem.with_suffix('.command.json').write_text(json.dumps(record,indent=2)+'\n')
    stem.with_suffix('.exit').write_text(str(result.returncode)+'\n')
    print(stage,name,'exit',result.returncode,'lines',record['lines'],flush=True)
    return record

def stage(name):
    for check, argv in COMMANDS:
        run(name,check,argv)
    if name == 'candidate':
        run(name,'strict-frontend',['node','tools/frontend/check-frontend-style.mjs','--mode','strict-full',*[a for p in FRONTEND for a in ['--file',p]]])
    root = Path('/workspace/FsusBlog-three-leaf' + ('-base' if name == 'base' else ''))
    config = root/'src/frontend/.tmp/typecheck/tsconfig.fsusui-local.app.generated.json'
    if config.exists():
        (OUT/(name+'-application-config.snapshot.json')).write_bytes(config.read_bytes())

with concurrent.futures.ThreadPoolExecutor(max_workers=2) as executor:
    list(executor.map(stage,['base','candidate']))
